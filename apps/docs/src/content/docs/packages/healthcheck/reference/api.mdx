The public API is intentionally split by runtime and integration. Import only from the package that owns the capability.

<ApiSymbol name="createHealthManager" importPath="@nexload-sdk/healthcheck" signature="createHealthManager(options: HealthManagerOptions): HealthManager" description="Creates a manager that registers checks and collectors, runs a scope, tracks shutdown state, and disposes runtime resources." />

<ApiSymbol name="defineHealthCheck" importPath="@nexload-sdk/healthcheck" signature="defineHealthCheck(definition: HealthCheckDefinition): HealthCheckDefinition" description="Defines a typed status-bearing check. Use it for custom probes." />

<ApiSymbol name="defineMetricCollector" importPath="@nexload-sdk/healthcheck" signature="defineMetricCollector(definition: MetricCollectorDefinition): MetricCollectorDefinition" description="Defines a monitoring-only collector that does not independently aggregate health status." />

<ApiSymbol name="shutdownCheck" importPath="@nexload-sdk/healthcheck" signature={'shutdownCheck(options?): HealthCheckDefinition<"shutdown">'} description="Reports unhealthy when manager shutdown state is set. It is critical for liveness and readiness by default." />

<ApiSymbol name="memoryCheck" importPath="@nexload-sdk/healthcheck" signature={'memoryCheck(options?): HealthCheckDefinition<"memory">'} description="Evaluates runtime memory snapshots with availability or used-ratio thresholds." />

<ApiSymbol name="httpCheck" importPath="@nexload-sdk/healthcheck" signature="httpCheck(name, url, options?): HealthCheckDefinition" description="Runs an abortable HTTP dependency check with expected-status and optional body validation." />

<ApiSymbol name="toHealthJson" importPath="@nexload-sdk/healthcheck" signature="toHealthJson(report, options?): HealthReport" description="Sanitizes a report with explicit detail and redaction options. Manager profiles are context hints, not serializer filters." />

<ApiSymbol name="stringifyHealthJson" importPath="@nexload-sdk/healthcheck" signature="stringifyHealthJson(report, options?): string" description="Returns JSON text for a serialized health report." />
